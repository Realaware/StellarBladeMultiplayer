cmake_minimum_required(VERSION 3.24)
project(StellarBladeCoop VERSION 0.1.0 LANGUAGES CXX)

include(CTest)
find_package(Threads REQUIRED)

add_library(sbcoop_core
    src/protocol/codec.cpp
    src/replication/entity_registry.cpp
    src/replication/snapshot_buffer.cpp
    src/save/save_system.cpp
    src/save/file_flag_store.cpp
    src/save/backup_result.cpp
    src/save/runtime_controller.cpp
)
target_include_directories(sbcoop_core PUBLIC include)
target_compile_features(sbcoop_core PUBLIC cxx_std_20)
target_link_libraries(sbcoop_core PUBLIC Threads::Threads)
set_target_properties(sbcoop_core PROPERTIES CXX_EXTENSIONS OFF)

function(sbcoop_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endfunction()
sbcoop_warnings(sbcoop_core)

add_executable(sbcoop_harness tools/harness/main.cpp)
target_link_libraries(sbcoop_harness PRIVATE sbcoop_core Threads::Threads)
sbcoop_warnings(sbcoop_harness)

if(BUILD_TESTING)
    add_executable(sbcoop_tests tests/core_tests.cpp)
    target_link_libraries(sbcoop_tests PRIVATE sbcoop_core Threads::Threads)
    sbcoop_warnings(sbcoop_tests)
    add_test(NAME core COMMAND sbcoop_tests)
    add_test(NAME harness_clean COMMAND sbcoop_harness)
    add_test(NAME harness_impaired COMMAND sbcoop_harness --impaired)
    add_executable(sbcoop_save_tests tests/save_tests.cpp)
    target_link_libraries(sbcoop_save_tests PRIVATE sbcoop_core)
    sbcoop_warnings(sbcoop_save_tests)
    add_test(NAME save_system COMMAND sbcoop_save_tests)
    set_tests_properties(core harness_clean harness_impaired save_system PROPERTIES TIMEOUT 30)
    add_executable(sbcoop_backup_result_tests tests/backup_result_tests.cpp)
    target_link_libraries(sbcoop_backup_result_tests PRIVATE sbcoop_core)
    sbcoop_warnings(sbcoop_backup_result_tests)
    add_test(NAME backup_result COMMAND sbcoop_backup_result_tests)
    set_tests_properties(backup_result PROPERTIES TIMEOUT 30)
    add_executable(sbcoop_save_runtime_tests tests/save_runtime_tests.cpp)
    target_link_libraries(sbcoop_save_runtime_tests PRIVATE sbcoop_core)
    sbcoop_warnings(sbcoop_save_runtime_tests)
    add_test(NAME save_runtime COMMAND sbcoop_save_runtime_tests)
    set_tests_properties(save_runtime PROPERTIES TIMEOUT 30)
endif()

    target_link_libraries(sbcoop_backup_probe PRIVATE sbcoop_backup_runtime)

    if(BUILD_TESTING)

option(SBCOOP_BUILD_SAVE_UI "Build the Dear ImGui save UI and its preview" OFF)
set(SBCOOP_IMGUI_VERSION "1.91.9b" CACHE STRING "Pinned Dear ImGui version for the standalone panel")
set_property(CACHE SBCOOP_IMGUI_VERSION PROPERTY STRINGS "1.91.9b" "1.92.1")
set(SBCOOP_IMGUI_DIR "${CMAKE_SOURCE_DIR}/out/dependencies/imgui-v${SBCOOP_IMGUI_VERSION}/imgui-${SBCOOP_IMGUI_VERSION}" CACHE PATH "Pinned Dear ImGui source directory")
if(SBCOOP_BUILD_SAVE_UI)
    set(sbcoop_imgui_versions "1.91.9b" "1.92.1")
    if(NOT SBCOOP_IMGUI_VERSION IN_LIST sbcoop_imgui_versions)
        message(FATAL_ERROR "Unsupported Dear ImGui version. Use the pinned 1.91.9b or 1.92.1 source.")
    endif()
    if(NOT EXISTS "${SBCOOP_IMGUI_DIR}/imgui.cpp")
        message(FATAL_ERROR "Dear ImGui is missing. Run tools/fetch-save-ui.ps1 first.")
    endif()
    file(STRINGS "${SBCOOP_IMGUI_DIR}/imgui.h" imgui_version_line REGEX "^#define IMGUI_VERSION ")
    string(REGEX MATCH "\"([^\"]+)\"" imgui_version_match "${imgui_version_line}")
    if(NOT CMAKE_MATCH_1 STREQUAL SBCOOP_IMGUI_VERSION)
        message(FATAL_ERROR "This UI profile requires Dear ImGui ${SBCOOP_IMGUI_VERSION}. Set SBCOOP_IMGUI_DIR to its matching pinned source directory.")
    endif()
    add_library(sbcoop_imgui STATIC
        "${SBCOOP_IMGUI_DIR}/imgui.cpp" "${SBCOOP_IMGUI_DIR}/imgui_draw.cpp"
        "${SBCOOP_IMGUI_DIR}/imgui_tables.cpp" "${SBCOOP_IMGUI_DIR}/imgui_widgets.cpp")
    target_include_directories(sbcoop_imgui SYSTEM PUBLIC "${SBCOOP_IMGUI_DIR}")
    add_library(sbcoop_save_ui src/ui/save_panel.cpp)
    target_link_libraries(sbcoop_save_ui PUBLIC sbcoop_core sbcoop_imgui)
    sbcoop_warnings(sbcoop_save_ui)
    if(WIN32)
        add_executable(sbcoop_save_preview WIN32 tools/save-preview/main.cpp
            "${SBCOOP_IMGUI_DIR}/backends/imgui_impl_win32.cpp"
            "${SBCOOP_IMGUI_DIR}/backends/imgui_impl_dx11.cpp")
        target_link_libraries(sbcoop_save_preview PRIVATE sbcoop_save_ui d3d11 dxgi bcrypt)
        target_include_directories(sbcoop_save_preview SYSTEM PRIVATE "${SBCOOP_IMGUI_DIR}/backends")
        target_compile_features(sbcoop_save_preview PRIVATE cxx_std_20)
        target_compile_definitions(sbcoop_save_preview PRIVATE UNICODE _UNICODE)
        target_compile_options(sbcoop_save_preview PRIVATE /utf-8)
    endif()
    if(BUILD_TESTING)
        add_executable(sbcoop_save_ui_tests tests/save_ui_tests.cpp)
        target_link_libraries(sbcoop_save_ui_tests PRIVATE sbcoop_save_ui)
        sbcoop_warnings(sbcoop_save_ui_tests)
        add_test(NAME save_ui COMMAND sbcoop_save_ui_tests)
        set_tests_properties(save_ui PROPERTIES TIMEOUT 30)
    endif()
endif()
